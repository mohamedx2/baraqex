💻 Simple Code Example


// One file, full-stack app!
import { jsx, useState, callWasmFunction } from 'baraqex';

function SuperApp() {
  const [result, setResult] = useState(0);
  
  const handleCalculate = async () => {
    // This runs Go code at near-native speed!
    const answer = callWasmFunction('fibonacci', 1000000);
    setResult(answer);
  };
  
  return (
    <div>
      <h1>🚀 Baraqex App</h1>
      <button onClick={handleCalculate}>
        Calculate Fibonacci(1M) ⚡
      </button>
      <p>Result: {result}</p>
    </div>
  );
}

// SSR happens automatically! 🎉

That's it! No React, no complex setup, just pure performance. 🔥
